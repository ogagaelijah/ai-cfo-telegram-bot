const {
    getSession,
    setSession,
    clearSession
} = require("../../states/sessionManager");

const STATES =
    require("../../constants/states");

const accountContext =
    require("../../services/accountContext");

const personalDebtorApplication =
    require("../../application/personal/personalDebtors");

const personalDebtorKeyboard =
    require("../../keyboards/personal/personalDebtorKeyboard");


// ======================================================
// PERSONAL DEBTORS TELEGRAM FLOW
// ======================================================
//
// INTERFACE ADAPTER ONLY
//
// This file is Telegram-specific.
//
// It is responsible for:
//
// - Telegram input
// - Telegram sessions
// - Telegram messages
// - Telegram keyboards
// - Resolving Telegram user -> current account
//
// It does NOT contain:
//
// - Database queries
// - Financial persistence
// - Core debtor business logic
// - Account authorization logic
// - Repository logic
//
// Architecture:
//
// Telegram
//    ↓
// This Flow
//    ↓
// Personal Debtors Application
//    ↓
// Personal Debtor Service
//    ↓
// Personal Debtor Repository
//    ↓
// SQLite
//
// ======================================================


// ======================================================
// FORMAT MONEY
// ======================================================

function money(amount) {

    return Number(
        amount || 0
    ).toLocaleString(
        "en-NG"
    );

}


// ======================================================
// GET ACCOUNT ID
// ======================================================

function getAccountId(account) {

    return (
        account.accountId ||
        account.id
    );

}


// ======================================================
// GET ORIGINAL AMOUNT
// ======================================================

function getOriginalAmount(debtor) {

    return Number(
        debtor.original_amount ??
        debtor.originalAmount ??
        0
    );

}


// ======================================================
// GET PAID AMOUNT
// ======================================================

function getPaidAmount(debtor) {

    return Number(
        debtor.paid_amount ??
        debtor.paidAmount ??
        0
    );

}


// ======================================================
// GET REMAINING AMOUNT
// ======================================================

function getRemaining(debtor) {

    return Number(
        debtor.remaining_amount ??
        debtor.remainingAmount ??
        0
    );

}


// ======================================================
// GET DUE DATE
// ======================================================

function getDueDate(debtor) {

    return (
        debtor.due_date ??
        debtor.dueDate ??
        null
    );

}


// ======================================================
// GET NOTES
// ======================================================

function getNotes(debtor) {

    return (
        debtor.notes ||
        ""
    );

}


// ======================================================
// VALIDATE INTEGER ID
// ======================================================

function parsePositiveInteger(value) {

    const number =
        Number(value);


    if (
        !Number.isInteger(number) ||
        number <= 0
    ) {

        return null;

    }


    return number;

}


// ======================================================
// PARSE MONEY INPUT
// ======================================================

function parseAmount(value) {

    const amount =
        Number(
            String(value)
                .replace(/,/g, "")
                .trim()
        );


    if (
        !Number.isFinite(amount) ||
        amount <= 0
    ) {

        return null;

    }


    return amount;

}


// ======================================================
// MAIN FLOW
// ======================================================

module.exports = async function personalDebtorsFlow(ctx) {

    const telegramId =
        ctx.from &&
        ctx.from.id
            ? ctx.from.id
            : null;


    if (!telegramId) {

        return true;

    }


    try {

        // ==================================================
        // GET SESSION
        // ==================================================

        const session =
            getSession(
                telegramId
            );


        if (!session) {

            return false;

        }


        // ==================================================
        // RESOLVE CURRENT ACCOUNT
        // ==================================================
        //
        // Telegram is converted into the user's active
        // account here.
        //
        // Everything below works with accountId.
        //
        // ==================================================

        const account =
            accountContext.getCurrentAccount(
                telegramId
            );


        if (!account) {

            clearSession(
                telegramId
            );


            await ctx.reply(
                "⚠️ No active account was found. Please select an account first."
            );


            return true;

        }


        const accountId =
            getAccountId(
                account
            );


        if (!accountId) {

            clearSession(
                telegramId
            );


            await ctx.reply(
                "⚠️ Your current account could not be identified."
            );


            return true;

        }


        // ==================================================
        // GET TELEGRAM TEXT
        // ==================================================

        const text =
            ctx.message &&
            ctx.message.text
                ? ctx.message.text.trim()
                : "";


        // ==================================================
        // ADD DEBTOR - NAME
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_NAME
        ) {

            if (!text) {

                await ctx.reply(
                    "Please enter the debtor's name."
                );


                return true;

            }


            setSession(

                telegramId,

                {

                    state:
                        STATES.WAITING_FOR_PERSONAL_DEBTOR_AMOUNT,

                    data: {

                        name:
                            text

                    }

                }

            );


            await ctx.reply(

                "💰 How much does this person owe you?\n\n" +
                "Example: 50000"

            );


            return true;

        }


        // ==================================================
        // ADD DEBTOR - AMOUNT
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_AMOUNT
        ) {

            const amount =
                parseAmount(
                    text
                );


            if (!amount) {

                await ctx.reply(

                    "⚠️ Please enter a valid positive amount.\n\n" +
                    "Example: 50000"

                );


                return true;

            }


            setSession(

                telegramId,

                {

                    state:
                        STATES.WAITING_FOR_PERSONAL_DEBTOR_NOTE,

                    data: {

                        ...(session.data || {}),

                        originalAmount:
                            amount

                    }

                }

            );


            await ctx.reply(

                "📝 Enter a note for this debtor.\n\n" +
                "Type NONE if you do not want to add a note."

            );


            return true;

        }


        // ==================================================
        // ADD DEBTOR - NOTE
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_NOTE
        ) {

            const data =
                session.data || {};


            const notes =
                text.toUpperCase() === "NONE"
                    ? ""
                    : text;


            const debtor =
                personalDebtorApplication.createPersonalDebtor(

                    accountId,

                    {

                        name:
                            data.name,

                        originalAmount:
                            data.originalAmount,

                        notes

                    }

                );


            clearSession(
                telegramId
            );


            await ctx.reply(

                "✅ DEBTOR ADDED\n\n" +

                `👤 ${debtor.name}\n` +

                `💰 Total Owed: ₦${money(
                    getOriginalAmount(debtor)
                )}\n` +

                `💵 Remaining: ₦${money(
                    getRemaining(debtor)
                )}\n` +

                `📌 Status: ${debtor.status}`,

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // VIEW DEBTORS
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_VIEW
        ) {

            const debtors =
                personalDebtorApplication.getPersonalDebtors(
                    accountId
                );


            clearSession(
                telegramId
            );


            if (
                !debtors ||
                debtors.length === 0
            ) {

                await ctx.reply(

                    "📋 DEBTORS\n\n" +
                    "You currently have no debtors.",

                    personalDebtorKeyboard

                );


                return true;

            }


            let message =
                "📋 YOUR DEBTORS\n\n";


            debtors.forEach(

                (debtor, index) => {

                    message +=

                        `${index + 1}. 👤 ${debtor.name}\n` +

                        `   ID: ${debtor.id}\n` +

                        `   💰 Total: ₦${money(
                            getOriginalAmount(debtor)
                        )}\n` +

                        `   💵 Paid: ₦${money(
                            getPaidAmount(debtor)
                        )}\n` +

                        `   🔴 Remaining: ₦${money(
                            getRemaining(debtor)
                        )}\n` +

                        `   📌 Status: ${debtor.status}\n`;


                    const dueDate =
                        getDueDate(
                            debtor
                        );


                    if (dueDate) {

                        message +=
                            `   📅 Due: ${dueDate}\n`;

                    }


                    const notes =
                        getNotes(
                            debtor
                        );


                    if (notes) {

                        message +=
                            `   📝 Note: ${notes}\n`;

                    }


                    message += "\n";

                }

            );


            await ctx.reply(

                message,

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // RECEIVE PAYMENT - DEBTOR ID
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_PAYMENT_DEBT
        ) {

            const debtorId =
                parsePositiveInteger(
                    text
                );


            if (!debtorId) {

                await ctx.reply(
                    "⚠️ Please enter a valid debtor ID."
                );


                return true;

            }


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                return true;

            }


            const remaining =
                getRemaining(
                    debtor
                );


            if (remaining <= 0) {

                await ctx.reply(

                    "ℹ️ This debtor has already been fully paid.",

                    personalDebtorKeyboard

                );


                clearSession(
                    telegramId
                );


                return true;

            }


            setSession(

                telegramId,

                {

                    state:
                        STATES.WAITING_FOR_PERSONAL_DEBTOR_PAYMENT_AMOUNT,

                    data: {

                        debtorId

                    }

                }

            );


            await ctx.reply(

                `💳 ${debtor.name} currently owes you ₦${money(remaining)}.\n\n` +
                "How much did they pay?"

            );


            return true;

        }


        // ==================================================
        // RECEIVE PAYMENT - AMOUNT
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_PAYMENT_AMOUNT
        ) {

            const amount =
                parseAmount(
                    text
                );


            if (!amount) {

                await ctx.reply(
                    "⚠️ Please enter a valid positive payment amount."
                );


                return true;

            }


            const debtorId =
                session.data &&
                session.data.debtorId;


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                clearSession(
                    telegramId
                );


                return true;

            }


            const remaining =
                getRemaining(
                    debtor
                );


            if (amount > remaining) {

                await ctx.reply(

                    `⚠️ Payment cannot be greater than the remaining balance of ₦${money(remaining)}.`

                );


                return true;

            }


            const updatedDebtor =
                personalDebtorApplication.recordPersonalDebtorPayment(

                    accountId,

                    debtorId,

                    amount

                );


            clearSession(
                telegramId
            );


            await ctx.reply(

                "✅ PAYMENT RECEIVED\n\n" +

                `👤 ${updatedDebtor.name}\n` +

                `💵 Payment: ₦${money(amount)}\n` +

                `💰 Remaining: ₦${money(
                    getRemaining(updatedDebtor)
                )}\n` +

                `📌 Status: ${updatedDebtor.status}`,

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // DEBTOR SUMMARY
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_SUMMARY
        ) {

            const summary =
                personalDebtorApplication.getSummary(
                    accountId
                );


            clearSession(
                telegramId
            );


            const totalDebtors =
                Number(
                    summary.total_debtors || 0
                );


            const totalOwed =
                Number(
                    summary.total_original_amount || 0
                );


            const totalPaid =
                Number(
                    summary.total_paid_amount || 0
                );


            const totalRemaining =
                Number(
                    summary.total_remaining_amount || 0
                );


            await ctx.reply(

                "📈 DEBTOR SUMMARY\n\n" +

                `👥 Total Debtors: ${totalDebtors}\n` +

                `💰 Total Owed: ₦${money(totalOwed)}\n` +

                `💵 Total Received: ₦${money(totalPaid)}\n` +

                `🔴 Total Outstanding: ₦${money(totalRemaining)}\n\n` +

                `🟢 Active: ${Number(
                    summary.active_debtors || 0
                )}\n` +

                `✅ Paid: ${Number(
                    summary.paid_debtors || 0
                )}`,

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // UPDATE DEBTOR - ID
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_UPDATE_DEBT
        ) {

            const debtorId =
                parsePositiveInteger(
                    text
                );


            if (!debtorId) {

                await ctx.reply(
                    "⚠️ Please enter a valid debtor ID."
                );


                return true;

            }


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                return true;

            }


            setSession(

                telegramId,

                {

                    state:
                        STATES.WAITING_FOR_PERSONAL_DEBTOR_UPDATE_FIELD,

                    data: {

                        debtorId

                    }

                }

            );


            await ctx.reply(

                "✏️ UPDATE DEBTOR\n\n" +

                `👤 ${debtor.name}\n` +

                `💰 Total: ₦${money(
                    getOriginalAmount(debtor)
                )}\n` +

                `💵 Paid: ₦${money(
                    getPaidAmount(debtor)
                )}\n` +

                `🔴 Remaining: ₦${money(
                    getRemaining(debtor)
                )}\n\n` +

                "What would you like to update?\n\n" +

                "1. Name\n" +
                "2. Note\n\n" +

                "Please reply with 1 or 2."

            );


            return true;

        }


        // ==================================================
        // UPDATE DEBTOR - FIELD
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_UPDATE_FIELD
        ) {

            if (
                !["1", "2"].includes(text)
            ) {

                await ctx.reply(
                    "⚠️ Please reply with 1 or 2."
                );


                return true;

            }


            setSession(

                telegramId,

                {

                    state:
                        STATES.WAITING_FOR_PERSONAL_DEBTOR_UPDATE_VALUE,

                    data: {

                        ...(session.data || {}),

                        field:
                            text

                    }

                }

            );


            if (text === "1") {

                await ctx.reply(
                    "Enter the new debtor name."
                );

            }

            else {

                await ctx.reply(

                    "Enter the new note.\n\n" +
                    "Type NONE to clear the note."

                );

            }


            return true;

        }


        // ==================================================
        // UPDATE DEBTOR - VALUE
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_UPDATE_VALUE
        ) {

            const debtorId =
                session.data &&
                session.data.debtorId;


            const field =
                session.data &&
                session.data.field;


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                clearSession(
                    telegramId
                );


                return true;

            }


            const updateData = {};


            // --------------------------------------------------
            // NAME
            // --------------------------------------------------

            if (field === "1") {

                if (!text) {

                    await ctx.reply(
                        "Please enter a valid name."
                    );


                    return true;

                }


                updateData.name =
                    text;

            }


            // --------------------------------------------------
            // NOTE
            // --------------------------------------------------

            else if (field === "2") {

                updateData.notes =
                    text.toUpperCase() === "NONE"
                        ? ""
                        : text;

            }


            else {

                await ctx.reply(
                    "⚠️ Invalid update option."
                );


                clearSession(
                    telegramId
                );


                return true;

            }


            const updatedDebtor =
                personalDebtorApplication.updatePersonalDebtor(

                    accountId,

                    debtorId,

                    updateData

                );


            clearSession(
                telegramId
            );


            await ctx.reply(

                "✅ DEBTOR UPDATED\n\n" +

                `👤 ${updatedDebtor.name}\n` +

                `💰 Total: ₦${money(
                    getOriginalAmount(updatedDebtor)
                )}\n` +

                `💵 Paid: ₦${money(
                    getPaidAmount(updatedDebtor)
                )}\n` +

                `🔴 Remaining: ₦${money(
                    getRemaining(updatedDebtor)
                )}\n` +

                `📌 Status: ${updatedDebtor.status}`,

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // COMPLETE DEBTOR
        // ==================================================
        //
        // IMPORTANT:
        //
        // Completion is only allowed when the balance is
        // already zero.
        //
        // We do NOT invent a separate completeDebtor()
        // application method.
        //
        // The actual financial event is the final payment.
        //
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_COMPLETE
        ) {

            const debtorId =
                parsePositiveInteger(
                    text
                );


            if (!debtorId) {

                await ctx.reply(
                    "⚠️ Please enter a valid debtor ID."
                );


                return true;

            }


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                return true;

            }


            const remaining =
                getRemaining(
                    debtor
                );


            if (remaining > 0) {

                await ctx.reply(

                    `⚠️ This debtor still has an outstanding balance of ₦${money(remaining)}.\n\n` +
                    "Record the final payment before completing the debtor."

                );


                return true;

            }


            if (
                debtor.status !== "PAID"
            ) {

                personalDebtorApplication.updatePersonalDebtorStatus(

                    accountId,

                    debtorId,

                    "PAID"

                );

            }


            clearSession(
                telegramId
            );


            await ctx.reply(

                "✅ DEBTOR COMPLETED\n\n" +

                `👤 ${debtor.name}\n` +

                `💰 Amount: ₦${money(
                    getOriginalAmount(debtor)
                )}\n\n` +

                "The debtor has been marked as fully paid.",

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // DELETE DEBTOR
        // ==================================================

        if (
            session.state ===
            STATES.WAITING_FOR_PERSONAL_DEBTOR_DELETE
        ) {

            const debtorId =
                parsePositiveInteger(
                    text
                );


            if (!debtorId) {

                await ctx.reply(
                    "⚠️ Please enter a valid debtor ID."
                );


                return true;

            }


            const debtor =
                personalDebtorApplication.getPersonalDebtor(

                    accountId,

                    debtorId

                );


            if (!debtor) {

                await ctx.reply(
                    "⚠️ Debtor not found."
                );


                return true;

            }


            const result =
                personalDebtorApplication.deletePersonalDebtor(

                    accountId,

                    debtorId

                );


            if (
                !result ||
                !result.deleted
            ) {

                await ctx.reply(

                    "⚠️ The debtor could not be deleted."

                );


                return true;

            }


            clearSession(
                telegramId
            );


            await ctx.reply(

                "🗑️ DEBTOR DELETED\n\n" +

                `👤 ${debtor.name}\n\n` +

                "The debtor record has been removed.",

                personalDebtorKeyboard

            );


            return true;

        }


        // ==================================================
        // UNKNOWN SESSION
        // ==================================================

        return false;

    }

    catch (error) {

        console.error(
            "Personal debtors flow error:",
            error
        );


        clearSession(
            telegramId
        );


        await ctx.reply(

            "⚠️ Something went wrong while processing the debtor operation.\n\n" +
            "Please try again.",

            personalDebtorKeyboard

        );


        return true;

    }

};
